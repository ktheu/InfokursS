\question[3]
Erstelle eine leere Liste \texttt{a}. \\
Füge in \texttt{a} nacheinander die Zahlen ein: 6, 2, 17 \\
Gib die Liste aus. \\
Gib die Länge von \texttt{a} aus.
 


\begin{solutionbox}{3cm}
\begin{lstlisting}
a = []
a.append(6)
a.append(2)
a.append(17)
print(a)
print(len(a))
\end{lstlisting}
\end{solutionbox}
