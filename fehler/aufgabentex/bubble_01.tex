\begin{lstlisting}
def bubble_sort(a):
    getauscht = True
    while getauscht:
        getauscht = ____
        for i in range(len(a)-1):
            if a[i] > a[i+1]:
                a[i], a[i+1] = a[i+1], a[i]
                ____
\end{lstlisting}

   (wird zu Beginn jedes Durchgangs zurückgesetzt)

2. Stelle: getauscht = True
