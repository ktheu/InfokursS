\question[3]
Führe die Funktion von Hand aus. Welche Ausgabe erscheint auf der Konsole?
\begin{lstlisting}
def g(node):
    if node is None:
        return
    g(node.left)
    g(node.right)
    print(node.val, end=' ')

baum = Node(1, Node(2, None, Node(3)), Node(4, Node(5), Node(6)))
g(baum)
\end{lstlisting}
\begin{solutionbox}{2cm}
\texttt{3 2 5 6 4 1}
\end{solutionbox}
