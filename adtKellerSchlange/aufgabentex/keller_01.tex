\question[2]
a. Welche Datenstruktur wird hier verwendet (Liste/Keller/Schlange)? \\
b. Was erscheint auf der Konsole?
 
\begin{lstlisting}
a = []
for i in range(5):
    a.append(i)

print(a.pop(), end=' ')
print(a.pop(), end=' ')

for i in range(6,9):
    a.append(i)

while a:
    print(a.pop(), end=' ')
\end{lstlisting}

\begin{solutionbox}{2cm}
 
a. Keller ~~~ b. \texttt{4 3 8 7 6 2 1 0}
\end{solutionbox}
