\question[3]
a. Welche Datenstruktur wird hier verwendet (Liste/Keller/Schlange)? \\
b. Was erscheint auf der Konsole?
 
\begin{lstlisting}
from collections import deque
a = deque([]) 

for i in range(3):
    for j in range(2):
        a.append(j)
    a.append(i)

while a:
    print(a.popleft(), end = ' ')

\end{lstlisting}
