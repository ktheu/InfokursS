\begin{lstlisting}
def entferne_negative(a):
    for i in range(len(a)-1, -1, -1):
        if a[i] < 0:
            a.pop(i)
\end{lstlisting}

    verändert a: alle negativen Zahlen werden entfernt
    returns: None

    >>> b = [1, -2, 3]
    >>> entferne_negative(b)
    >>> b
    [1, 3]
    '''
