\question[3]
Schreibe die folgende Funktion als Pseudocode.
\begin{lstlisting}
def suche(a, x):
    for i in range(len(a)):
        if a[i] == x:
            return i
    return -1
\end{lstlisting}
\begin{solutionbox}{3.25cm}
\begin{lstlisting}[language={}]
algorithmus suche(a, x):
    für i von 0 bis len(a)-1:
        wenn a[i] == x:
            gib i zurück
    gib -1 zurück
\end{lstlisting}
\end{solutionbox}
