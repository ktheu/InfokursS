\question[3]
Der folgende Algorithmus gehört zu einer Funktion \texttt{ist\_sortiert(a)}.
Schreibe die Funktion in Python.

\includegraphics[height=3.4cm]{\pfad/nsd_python_01.png}

\begin{solutionbox}{3.25cm}
\begin{lstlisting}
def ist_sortiert(a):
    for i in range(len(a)-1):
        if a[i] > a[i+1]:
            return False
    return True
\end{lstlisting}
\end{solutionbox}
